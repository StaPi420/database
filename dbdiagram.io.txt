Table Authors { 
  id_author int [pk, increment] 
  last_name varchar(50) [not null] 
  first_name varchar(50) [not null] 
  middle_name varchar(50) 
  country varchar(50) 
  birth_date date 
  biography text 
} 

Table Genres { 
  id_genre int [pk, increment] 
  name varchar(100) [not null] 
  description text 
  bbk_code varchar(20) 
} 

Table Publishers { 
  id_publisher int [pk, increment] 
  name varchar(100) [not null] 
  city varchar(50) 
  phone varchar(20) 
  email varchar(50) 
  website varchar(100) 
} 

Table Reader_Categories { 
  id_category int [pk, increment] 
  name varchar(50) [not null] 
  max_books int [default: 5] 
  loan_days int [default: 14] 
  penalty_per_day decimal(10,2) [default: 0] 
} 

Table Copy_Statuses { 
  id_status int [pk, increment] 
  name varchar(50) [not null] 
  description text 
  available_for_loan boolean [default: false] 
} 

Table ILL_Types { 
  id_type int [pk, increment] 
  name varchar(50) [not null] 
  days int 
  cost decimal(10,2) 
} 

Table Employees { 
  id_employee int [pk, increment] 
  last_name varchar(50) [not null] 
  first_name varchar(50) [not null] 
  middle_name varchar(50) 
  position varchar(50) 
  phone varchar(20) 
  hire_date date 
} 

Table Readers { 
  id_reader int [pk, increment] 
  last_name varchar(50) [not null] 
  first_name varchar(50) [not null] 
  middle_name varchar(50) 
  birth_date date 
  phone varchar(20) 
  email varchar(50) 
  id_category int [ref: > Reader_Categories.id_category] 
  registration_date date
  blocked boolean [default: false] 
} 

Table Partner_Libraries { 
  id_library int [pk, increment] 
  name varchar(200) [not null] 
  city varchar(50) 
  contacts varchar(200) 
} 

Table Books { 
  id_book int [pk, increment] 
  title varchar(200) [not null] 
  isbn varchar(20) 
  publication_year int 
  page_count int 
  price decimal(10,2) 
  id_genre int [ref: > Genres.id_genre] 
  id_publisher int [ref: > Publishers.id_publisher] 
  language varchar(30) 
  annotation text 
} 

Table Book_Author { 
  id_record int [pk, increment] 
  id_book int [ref: > Books.id_book] 
  id_author int [ref: > Authors.id_author] 
} 

Table Copies { 
  id_copy int [pk, increment] 
  inventory_number varchar(20) [unique, not null] 
  id_book int [ref: > Books.id_book] 
  id_status int [ref: > Copy_Statuses.id_status] 
  arrival_date date 
  storage_location varchar(50) 
} 

Table Acquisitions { 
  id_acquisition int [pk, increment] 
  acquisition_date date [not null] 
  id_publisher int [ref: > Publishers.id_publisher] 
  id_employee int [ref: > Employees.id_employee] 
  invoice_number varchar(30) 
  batch_amount decimal(12,2) 
  book_count int 
} 

Table Acquisition_Copy { 
  id_record int [pk, increment] 
  id_acquisition int [ref: > Acquisitions.id_acquisition] 
  id_copy int [ref: > Copies.id_copy] 
  quantity int 
  copy_price decimal(10,2) 
} 

Table Loans { 
  id_loan int [pk, increment] 
  id_reader int [ref: > Readers.id_reader] 
  id_copy int [ref: > Copies.id_copy] 
  id_employee int [ref: > Employees.id_employee] 
  loan_date date
  due_date date 
  actual_return_date date 
  renewals int [default: 0] 
  penalty_amount decimal(10,2) [default: 0] 
} 

Table Returns { 
  id_return int [pk, increment] 
  id_loan int [ref: > Loans.id_loan] 
  return_date date
  book_condition varchar(50) 
  overdue_days int 
} 

Table Debts { 
  id_debt int [pk, increment] 
  id_loan int [ref: > Loans.id_loan] 
  id_reader int [ref: > Readers.id_reader] 
  penalty_amount decimal(10,2) 
  accrual_date date
  payment_date date 
  paid boolean [default: false] 
} 

Table ILL_Orders { 
  id_order int [pk, increment] 
  id_reader int [ref: > Readers.id_reader] 
  id_type int [ref: > ILL_Types.id_type] 
  id_employee int [ref: > Employees.id_employee] 
  order_date date
  completion_date date 
  status varchar(30) 
  cost decimal(10,2) 
} 

Table ILL_Order_Book { 
  id_record int [pk, increment] 
  id_order int [ref: > ILL_Orders.id_order] 
  id_book int [ref: > Books.id_book] 
  id_library int [ref: > Partner_Libraries.id_library] 
  quantity int 
} 

Table Reservations { 
  id_reservation int [pk, increment] 
  id_reader int [ref: > Readers.id_reader] 
  id_book int [ref: > Books.id_book] 
  reservation_date date
  expiration_date date 
  status varchar(30) 
} 

Table Status_History { 
  id_record int [pk, increment] 
  id_copy int [ref: > Copies.id_copy] 
  id_status int [ref: > Copy_Statuses.id_status] 
  change_date timestamp
  id_employee int [ref: > Employees.id_employee] 
  comment text 
} 

TableGroup Reference_Tables { 
  Authors 
  Genres 
  Publishers 
  Reader_Categories 
  Copy_Statuses 
  ILL_Types 
  Employees 
  Readers 
  Partner_Libraries 
} 

TableGroup Catalog { 
  Books 
  Book_Author 
  Copies 
  Status_History 
}

  Acquisitions 
  Acquisition_Copy 
} 

TableGroup Circulation { 
  Loans 
  Returns

  Reservations 
}

}
