        int id_acquisition PK
        date acquisition_date

    }
    ACQUISITION_COPY {
        int id_record PK

        decimal(10,2) copy_price

        int id_loan PK

        int id_order PK

    RESERVATIONS {
        int id_reservation PK

    }
